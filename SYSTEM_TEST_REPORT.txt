<script>
async function loadMissions() {
  try {
    const response = await fetch("http://localhost:4000/api/missions");
    if (!response.ok) throw new Error("API " + response.status);

    const missions = await response.json();

    if (missions.length > 0) {
      const m = missions[missions.length - 1];

      const alt = document.getElementById("alt");
      const tAlt = document.getElementById("tAlt");
      const fuel = document.getElementById("fuel");
      const tFuel = document.getElementById("tFuel");
      const speed = document.getElementById("speed");
      const tSpeed = document.getElementById("tSpeed");

      if (alt) alt.textContent = Number(m.altitude).toLocaleString();
      if (tAlt) tAlt.textContent = Number(m.altitude).toLocaleString();
      if (fuel) fuel.textContent = Number(m.fuel).toFixed(2);
      if (tFuel) tFuel.textContent = Number(m.fuel).toFixed(2);
      if (speed) speed.textContent = m.velocity;
      if (tSpeed) tSpeed.textContent = m.velocity;
    }

    console.log("2060 Mission API connected");
  } catch (error) {
    console.log("Mission API connection pending:", error.message);
  }
}

loadMissions();
</script>

function login2060(){
